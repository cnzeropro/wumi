# Wumi v0.2「图表+多币种」实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 统计图表（五图）+ 多币种（汇率管理与外币记账快照换算），验收「月报数字与手算一致」。

**Architecture:** 本地 Room 唯一真相源不变；新增 currencies 表（含同步基因，Migration 1→2）；换算收敛在纯 JVM `FxConverter`；汇率自动拉取走 WorkManager 周期任务（可关）；图表全部按 `baseAmountMinor` 本位币聚合，复用 `observeRange` + Kotlin 分组。

**Tech Stack:** Kotlin 2.2 + Compose M3 + Room + Hilt + WorkManager（既有）；新增 Vico 2.x（柱状/折线图）+ Compose Canvas 自绘环形图；HttpURLConnection（拉汇率，不引 OkHttp）。

## Global Constraints

- 金额一律 `Long` 最小单位；汇率存字符串（BigDecimal 解析）；禁 Float/Double 存金额
- `baseAmountMinor = round(amountMinor × rateToBase, HALF_UP)`；换算唯一出口 `FxConverter`
- 历史账单不随汇率波动重算（当笔快照入 `exchangeRate`）
- Room schema 1→2 用正式 Migration，**禁止** fallbackToDestructiveMigration
- UI 全中文；应用名 `Wumi · 吾米记账`；品牌米金 `Color(0xFFC8A24B)`
- 同步基因五字段（id/createdAt/updatedAt/deletedAt/deviceId）用于 currencies 表；outbox 与业务表同事务写入
- 构建命令：`.\gradlew.bat`（timeout 600000ms）；测试 `:app:testDebugUnitTest`
- 提交信息 Conventional Commits 中文；先测试后实现（TDD 红→绿）
- Java 21 / Gradle 9.5.0 / AGP 8.13.0 / minSdk 35 / targetSdk 36 / compileSdk 36 不可变

```kotlin
fun parseToMinor(text: String, decimals: Int): Long? {
    val trimmed = text.trim()
    if (trimmed.isEmpty()) return null
    val negative = trimmed.startsWith("-")
    val body = if (negative) trimmed.removePrefix("-") else trimmed
    val minor = parseBody(body, decimals) ?: return null   // 原有解析体抽为私有 parseBody
    return if (negative) -minor else minor
}
```

- [ ] **Step 2: 实现**

```kotlin
/** 外币→本位币唯一换算出口（金额铁律 spec §5.3）。 */
object FxConverter {
    fun toBaseMinor(amountMinor: Long, rateToBase: String): Long {
        val rate = BigDecimal(rateToBase)
        require(rate > BigDecimal.ZERO) { "invalid fx rate: $rateToBase" }
        return BigDecimal(amountMinor).multiply(rate)
            .setScale(0, RoundingMode.HALF_UP).longValueExact()
    }
}
```

```kotlin
/** open.er-api.com 响应解析：只取白名单币种，非法值跳过。 */
object ErApiParser {
    fun parse(json: String, allowed: Set<String>): Map<String, String> = runCatching {
        val root = Json.parseToJsonElement(json).jsonObject
        if (root["result"]?.jsonPrimitive?.contentOrNull != "success") return emptyMap()
        val rates = root["rates"]?.jsonObject ?: return emptyMap()
        rates.mapNotNull { (code, v) ->
            if (code !in allowed) return@mapNotNull null
            val s = runCatching { v.jsonPrimitive.content }.getOrNull() ?: return@mapNotNull null
            val d = runCatching { BigDecimal(s) }.getOrNull() ?: return@mapNotNull null
            if (d <= BigDecimal.ZERO) return@mapNotNull null
            code to d.stripTrailingZeros().toPlainString()
        }.toMap()
    }.getOrDefault(emptyMap())
}
```

```kotlin
    fun fetch(allowed: Set<String>): Result<Map<String, String>> = runCatching {
        val conn = java.net.URI("https://open.er-api.com/v6/latest/CNY").toURL()

        conn.connectTimeout = 10_000; conn.readTimeout = 10_000; conn.requestMethod = "GET"
```

```kotlin
class CurrencyRepository(
    private val db: WumiDatabase,
    private val dao: CurrencyDao,
    private val outboxDao: OutboxDao,
    private val settingDao: SettingDao,
    @DeviceId private val deviceId: String,
    private val fetch: (Set<String>) -> Result<Map<String, String>> = FxRateFetcher::fetch,
) {
    fun observeAll() = dao.observeAll()
    suspend fun getByCode(code: String) = dao.getByCode(code)
    suspend fun seedIfEmpty() { /* COUNT=0 则 upsertAll(DefaultFx.defs(now, deviceId)) */ }
    suspend fun updateRate(code: String, rate: String): Boolean {
        val d = runCatching { BigDecimal(rate) }.getOrNull() ?: return false
        if (d <= BigDecimal.ZERO) return false
        val now = System.currentTimeMillis()
        db.withTransactionSafe {
            dao.updateRate(code, d.stripTrailingZeros().toPlainString(), now)
            outboxDao.enqueue(OutboxEntity("currency", code, now))
        }
        return true
    }
    suspend fun refreshAll(): Result<Int> = fetch(allCodes()).mapCatching { rates ->
        val now = System.currentTimeMillis()
        db.withTransactionSafe {
            rates.forEach { (code, r) -> dao.updateRate(code, r, now); outboxDao.enqueue(OutboxEntity("currency", code, now)) }
        }
        rates.size
    }
    suspend fun autoSyncEnabled(): Boolean = settingDao.get("autoFxRate") != "0"
    suspend fun setAutoSync(enabled: Boolean) = settingDao.put(SettingEntity("autoFxRate", if (enabled) "1" else "0"))
}
```

```kotlin
fun applyAutoSync(context: Context) {
    val wm = WorkManager.getInstance(context)
    // enabled 需在协程内读：调用方为 suspend 或用 runBlocking（Application onCreate 一次性）
    if (runBlocking { autoSyncEnabled() }) {

    }
}
```
